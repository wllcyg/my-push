## 条件语句

### if 语句基础

```python
# 基本 if 语句
age = 18
if age >= 18:
    print("你是成年人")

# if-else
score = 85
if score >= 60:
    print("及格")
else:
    print("不及格")

# if-elif-else（多分支）
score = 85
if score >= 90:
    grade = 'A'
elif score >= 80:
    grade = 'B'
elif score >= 70:
    grade = 'C'
elif score >= 60:
    grade = 'D'
else:
    grade = 'F'
print(f"成绩等级：{grade}")
```

### 条件表达式

#### 布尔值和真值测试
```python
# 假值（Falsy）：
# False, None, 0, 0.0, "", [], {}, set()

# 真值测试
if []:
    print("不会执行")

if [1, 2, 3]:
    print("会执行")

# 常见模式
name = input("Enter name: ")
if name:  # 检查是否为空字符串
    print(f"Hello, {name}")
else:
    print("Name cannot be empty")

# 数字
num = 0
if num:
    print("不会执行")
if num == 0:
    print("会执行")
```

#### 逻辑运算符
```python
# and、or、not
age = 20
has_license = True

# and：所有条件都为真
if age >= 18 and has_license:
    print("可以开车")

# or：至少一个条件为真
is_weekend = True
is_holiday = False
if is_weekend or is_holiday:
    print("可以休息")

# not：取反
is_raining = False
if not is_raining:
    print("不需要带伞")

# 组合使用
if (age >= 18 and has_license) or age >= 21:
    print("满足条件")

# 短路求值
def expensive_check():
    print("执行了昂贵的检查")
    return True

# expensive_check() 不会被调用
if False and expensive_check():
    pass

# expensive_check() 不会被调用
if True or expensive_check():
    pass
```

#### 比较链
```python
# Python 支持链式比较
x = 5
if 0 < x < 10:
    print("x 在 0 和 10 之间")

# 等价于
if 0 < x and x < 10:
    print("x 在 0 和 10 之间")

# 更多例子
score = 85
if 80 <= score <= 90:
    print("成绩良好")

# 可以链接多个比较
if 1 < x < 5 < y < 10:
    print("满足所有条件")
```

### 三元运算符（条件表达式）

```python
# 语法：value_if_true if condition else value_if_false

age = 20
status = "成年人" if age >= 18 else "未成年人"

# 嵌套三元运算符（不推荐，可读性差）
score = 85
grade = 'A' if score >= 90 else ('B' if score >= 80 else 'C')

# 更好的写法
if score >= 90:
    grade = 'A'
elif score >= 80:
    grade = 'B'
else:
    grade = 'C'

# 实际应用
numbers = [1, 2, 3, 4, 5]
result = [x * 2 if x % 2 == 0 else x for x in numbers]
# [1, 4, 3, 8, 5]
```

### match-case 语句（Python 3.10+）

```python
# 类似其他语言的 switch-case

def http_status(status):
    match status:
        case 200:
            return "OK"
        case 404:
            return "Not Found"
        case 500:
            return "Internal Server Error"
        case _:  # 默认情况
            return "Unknown"

# 模式匹配
def process_command(command):
    match command.split():
        case ["quit"]:
            print("退出程序")
        case ["load", filename]:
            print(f"加载文件：{filename}")
        case ["save", filename]:
            print(f"保存文件：{filename}")
        case _:
            print("未知命令")

# 更复杂的模式
def describe_point(point):
    match point:
        case (0, 0):
            print("原点")
        case (0, y):
            print(f"Y轴上的点：{y}")
        case (x, 0):
            print(f"X轴上的点：{x}")
        case (x, y):
            print(f"点：({x}, {y})")

# 使用 guards
def categorize_number(num):
    match num:
        case n if n < 0:
            print("负数")
        case 0:
            print("零")
        case n if n > 0:
            print("正数")
```

#### range() 函数
```python
# range(stop)
for i in range(5):
    print(i)  # 0, 1, 2, 3, 4

# range(start, stop)
for i in range(1, 6):
    print(i)  # 1, 2, 3, 4, 5

# range(start, stop, step)
for i in range(0, 10, 2):
    print(i)  # 0, 2, 4, 6, 8

# 反向
for i in range(10, 0, -1):
    print(i)  # 10, 9, 8, ..., 1

# 负数范围
for i in range(-5, 0):
    print(i)  # -5, -4, -3, -2, -1

# range 是惰性求值（节省内存）

```

#### enumerate() - 带索引遍历
```python
fruits = ["apple", "banana", "orange"]

# 获取索引和值
for index, fruit in enumerate(fruits):
    print(f"{index}: {fruit}")

# 1: banana

# 指定起始索引
```

### break 语句

```python

for i in range(10):
    if i == 5:

# i=1, j=0
# i=2, j=0
```

### continue 语句

```python
# continue：跳过本次迭代，继续下一次
for i in range(10):
    if i % 2 == 0:
        continue  # 跳过偶数
    print(i)
# 输出：1, 3, 5, 7, 9

# 跳过特定条件
words = ["hello", "world", "", "python", ""]
for word in words:

# 数据过滤
numbers = [1, -2, 3, -4, 5, -6]
positive_sum = 0
for num in numbers:
    if num < 0:
        continue
    positive_sum += num
print(positive_sum)  # 9
```

## 上下文管理器

### with 语句

```python
# 基本用法：文件操作
with open("file.txt", "r") as f:
    content = f.read()
# 文件自动关闭，即使发生异常

# 等价于
f = open("file.txt", "r")
try:
    content = f.read()
finally:
    f.close()

# 多个上下文管理器
with open("input.txt") as infile, open("output.txt", "w") as outfile:
    for line in infile:
        outfile.write(line.upper())
```

### 自定义上下文管理器

#### 使用类
```python
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None
    
    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()
        # 返回 True 表示抑制异常
        # 返回 False 或 None 表示传播异常
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello")
```

#### 使用装饰器
```python
from contextlib import contextmanager

@contextmanager
def file_manager(filename, mode):
    try:
        f = open(filename, mode)
        yield f
    finally:
        f.close()

# 使用
with file_manager("test.txt", "w") as f:
    f.write("Hello")
```

```python
from contextlib import contextmanager

@contextmanager
def change_dir(path):
    old_dir = os.getcwd()
    os.chdir(path)

        yield
    finally:
        os.chdir(old_dir)

with change_dir("/tmp"):
    # 在 /tmp 目录中操作

# 自动恢复原目录

def temporary_setting(name, value):
    old_value = getattr(config, name)
    setattr(config, name, value)
    try:
        yield
    finally:
        setattr(config, name, old_value)
```

### 案例 1：猜数字游戏（完整版）

```python
import random

def guess_number_game():
    """完整的猜数字游戏"""
    print("=== 猜数字游戏 ===")
    print("我想了一个 1-100 之间的数字")
    
    secret = random.randint(1, 100)
    attempts = 0
    max_attempts = 10
    
    while attempts < max_attempts:
        try:
            guess = int(input(f"\n第 {attempts + 1} 次猜测: "))
            
            if guess < 1 or guess > 100:
                print("请输入 1-100 之间的数字！")
                continue
            
            attempts += 1
            
            if guess < secret:
                print("太小了！")
            elif guess > secret:
                print("太大了！")
            else:
                print(f"🎉 恭喜你猜对了！")
                print(f"你用了 {attempts} 次")
                break
        
        except ValueError:
            print("请输入有效的数字！")
    else:
        print(f"\n游戏结束！答案是 {secret}")

guess_number_game()
```

### 案例 2：待办事项管理器

```python
def todo_manager():
    """待办事项管理器"""
    todos = []
    
    while True:
        print("\n=== 待办事项 ===")
        print("1. 查看任务")
        print("2. 添加任务")
        print("3. 完成任务")
        print("4. 删除任务")
        print("5. 退出")
        
        try:
            choice = input("\n请选择操作: ")
            
            if choice == '1':
                if not todos:
                    print("没有任务")
                else:
                    for i, task in enumerate(todos, 1):
                        status = "✓" if task['done'] else " "
                        print(f"{i}. [{status}] {task['title']}")
            
            elif choice == '2':
                title = input("任务标题: ")
                if title:
                    todos.append({'title': title, 'done': False})
                    print("任务已添加！")
                else:
                    print("标题不能为空")
            
            elif choice == '3':
                if not todos:
                    print("没有任务")
                    continue
                
                try:
                    index = int(input("完成哪个任务？: ")) - 1
                    if 0 <= index < len(todos):
                        todos[index]['done'] = True
                        print("任务已完成！")
                    else:
                        print("无效的任务编号")
                except ValueError:
                    print("请输入有效的数字")
            
            elif choice == '4':
                if not todos:
                    print("没有任务")
                    continue
                
                try:
                    index = int(input("删除哪个任务？: ")) - 1

                        removed = todos.pop(index)
                        print(f"已删除: {removed['title']}")
                    else:
                        print("无效的任务编号")
                except ValueError:
                    print("请输入有效的数字")
            
            elif choice == '5':
                print("再见！")
                break
            
            else:
                print("无效的选择")
        
        except KeyboardInterrupt:
            print("\n\n程序被中断")
            break

todo_manager()
```

```python
                line_number += 1
                
                if 'ERROR' in line:
                    # 提取错误类型
                    try:
                        error_type = line.split('ERROR:')[1].split()[0]
                        error_count[error_type] = error_count.get(error_type, 0) + 1
                    except IndexError:
                        print(f"第 {line_number} 行格式错误")
                        continue
    
    except FileNotFoundError:
        print(f"文件 {filename} 不存在")
        return None
    except PermissionError:
        print(f"没有权限读取文件 {filename}")
        return None
    except Exception as e:
        print(f"处理文件时发生错误: {e}")
        return None
    else:
        print(f"\n成功处理 {line_number} 行")
        return error_count
    finally:
        print("文件处理完成")

# 使用
result = process_log_file("app.log")
if result:
    print("\n错误统计:")
    for error_type, count in result.items():
        print(f"{error_type}: {count}")
```

```python
def get_validated_input():
    """获取并验证用户输入"""
    
    # 验证邮箱
    while True:
        email = input("输入邮箱: ")
        if '@' in email and '.' in email.split('@')[1]:
            break
        print("邮箱格式不正确")
    
    # 验证年龄
    while True:
        try:
            age = int(input("输入年龄: "))
            if 0 < age < 150:
                break
            print("年龄必须在 1-150 之间")
        except ValueError:
            print("请输入有效的数字")
    
    # 验证密码
    while True:
        password = input("输入密码（至少6位）: ")
        if len(password) >= 6:
            confirm = input("确认密码: ")
            if password == confirm:
                break
            print("两次密码不一致")
        else:
            print("密码至少6位")
    
    return {
        'email': email,
        'age': age,
        'password': password
    }

user_data = get_validated_input()
print(f"\n注册成功！邮箱: {user_data['email']}")
```
