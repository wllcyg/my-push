## 字典（dict）

### 创建字典

```python
# 字面量（最常用）
user = {"name": "Alice", "age": 25}

# dict() 构造
user = dict(name="Alice", age=25)
pairs = dict([("a", 1), ("b", 2)])      # 从键值对列表创建

# 空字典（注意：{} 是空字典，不是空集合）
empty = {}

# 从两个序列创建
keys = ["name", "age"]
values = ["Alice", 25]
user = dict(zip(keys, values))          # {'name': 'Alice', 'age': 25}

# 用同一个值初始化多个键
scores = dict.fromkeys(["math", "english"], 0)   # {'math': 0, 'english': 0}
```

### 访问、修改、删除

```python
user = {"name": "Alice", "age": 25}

# 访问
print(user["name"])            # Alice
# print(user["email"])         # ❌ KeyError: 'email'

# 安全访问：get，键不存在时返回 None 或指定的默认值
print(user.get("email"))               # None
print(user.get("email", "未填写"))      # 未填写

# 新增 / 修改
user["email"] = "alice@example.com"    # 新增
user["age"] = 26                       # 修改

# 删除
del user["email"]                      # 键不存在会 KeyError
age = user.pop("age")                  # 删除并返回值
age = user.pop("age", None)            # 键不存在时返回默认值，不报错
last = user.popitem()                  # 删除并返回最后插入的键值对

# 判断键是否存在（判断的是键，不是值）
print("name" in user)                  # True
print("Alice" in user)                 # False
```

**重点：`d[key]` 会抛 `KeyError`，`d.get(key)` 不会。**
这和 JS 里 `obj.key` 不存在返回 `undefined` 的行为不同，Python 更严格。

### 常用方法

```python
d = {"a": 1, "b": 2}

print(d.keys())        # dict_keys(['a', 'b'])
print(d.values())      # dict_values([1, 2])
print(d.items())       # dict_items([('a', 1), ('b', 2)])
print(len(d))          # 2

# setdefault：键不存在才设置，并返回当前值
d.setdefault("c", 3)   # 设置 c=3
d.setdefault("a", 99)  # a 已存在，保持 1

# update：合并，同名键被覆盖
d.update({"a": 100, "z": 26})
d.update(b=200)        # 也可以用关键字参数

# clear / copy
copy = d.copy()        # 浅拷贝
d.clear()
```

### 合并字典（Python 3.9+）

```python
a = {"x": 1, "y": 2}
b = {"y": 20, "z": 30}

# | 运算符：生成新字典，后面的覆盖前面的
merged = a | b                 # {'x': 1, 'y': 20, 'z': 30}

# |= 原地合并（相当于 update）
a |= b

# 解包合并（所有 3.5+ 版本都能用，类似 JS 的 {...a, ...b}）
merged = {**a, **b}
```

### 遍历字典

```python
user = {"name": "Alice", "age": 25, "city": "Shanghai"}

for key in user:                       # 默认遍历键
    print(key)

for value in user.values():
    print(value)

for key, value in user.items():        # 最常用
    print(f"{key} = {value}")
```

> 从 Python 3.7 起，字典**保持插入顺序**，这是语言保证，可以放心依赖。

### 字典推导式

```python
# {键表达式: 值表达式 for 变量 in 可迭代对象 if 条件}
squares = {n: n ** 2 for n in range(1, 6)}
# {1: 1, 2: 4, 3: 9, 4: 16, 5: 25}

# 过滤
scores = {"Alice": 90, "Bob": 55, "Carol": 78}
passed = {name: s for name, s in scores.items() if s >= 60}
# {'Alice': 90, 'Carol': 78}

# 键值互换（值必须可哈希且唯一）
inverted = {v: k for k, v in scores.items()}

# 批量转换值
upper = {k: v.upper() for k, v in {"a": "x", "b": "y"}.items()}
```

### 嵌套字典

```python
company = {
    "name": "ACME",
    "employees": [
        {"name": "Alice", "skills": ["python", "sql"]},
        {"name": "Bob", "skills": ["vue", "ts"]},
    ],
}

print(company["employees"][0]["skills"][1])    # sql

# 安全地取深层值（不用写一长串判断）
city = company.get("address", {}).get("city", "未知")
```

### 字典与 JSON 互转

```python
import json

data = {"name": "Alice", "tags": ["a", "b"], "active": True, "score": None}

# 字典 -> JSON 字符串
text = json.dumps(data, ensure_ascii=False, indent=2)
# ensure_ascii=False 让中文不被转成 \uXXXX

# JSON 字符串 -> 字典
back = json.loads(text)

# 注意类型映射：True -> true，None -> null
```

## 集合（set）与 frozenset

集合是**无序、元素唯一**的容器，底层是哈希表，所以判断"元素在不在"非常快（接近 O(1)）。

### 创建集合

```python
s = {1, 2, 3, 3, 2}          # {1, 2, 3}，自动去重
s = set([1, 2, 2, 3])        # 从列表创建
s = set("hello")             # {'h', 'e', 'l', 'o'}

empty = set()                # ⚠️ 空集合只能这样写，{} 是空字典
```

### 增删

```python
s = {1, 2, 3}

s.add(4)                     # 添加一个
s.update([5, 6], {7})        # 添加多个
s.remove(1)                  # 元素不存在会 KeyError
s.discard(100)               # 元素不存在也不报错
x = s.pop()                  # 随机删除并返回一个（集合无序）
```

### 集合运算

```python
a = {1, 2, 3, 4}
b = {3, 4, 5, 6}

print(a | b)    # 并集 {1, 2, 3, 4, 5, 6}     等价 a.union(b)
print(a & b)    # 交集 {3, 4}                 等价 a.intersection(b)
print(a - b)    # 差集 {1, 2}                 等价 a.difference(b)
print(a ^ b)    # 对称差集 {1, 2, 5, 6}        等价 a.symmetric_difference(b)

print({1, 2} <= a)      # 子集 True
print(a >= {1, 2})      # 超集 True
print(a.isdisjoint({9}))  # 是否没有交集 True
```

### 集合的典型用途

```python
# 1. 去重
ids = [3, 1, 2, 3, 1]
unique = list(set(ids))                 # 顺序不保证

# 去重并保持顺序（字典保序）
unique_ordered = list(dict.fromkeys(ids))   # [3, 1, 2]

# 2. 快速判断是否存在
allowed = {"admin", "editor"}
if role in allowed:                     # 比 list 查找快得多
    ...

# 3. 求两份名单的差异
old_ids = {1, 2, 3}
new_ids = {2, 3, 4}
added = new_ids - old_ids               # {4}
removed = old_ids - new_ids             # {1}

# 集合推导式
evens = {n for n in range(10) if n % 2 == 0}
```

### frozenset：不可变集合

```python
fs = frozenset([1, 2, 3])
# fs.add(4)                 # ❌ AttributeError

# 因为不可变，所以可以做字典的键，或放进另一个集合
d = {frozenset({1, 2}): "pair"}
```

## 可哈希（hashable）是什么

字典的键、集合的元素，都必须是**可哈希**的：对象生命周期内哈希值不变。

```python
# ✅ 可哈希：int, float, str, bool, None, tuple（内部元素也都可哈希）, frozenset
d = {1: "a", "k": "b", (1, 2): "c"}

# ❌ 不可哈希：list, dict, set
# d = {[1, 2]: "x"}         # TypeError: unhashable type: 'list'
# s = {{"a": 1}}            # TypeError: unhashable type: 'dict'

# 元组里包含列表也不行
# {(1, [2]): "x"}           # TypeError
```

**经验法则：不可变对象基本可哈希，可变对象不可哈希。**

## collections 里的常用工具

### defaultdict：访问不存在的键时自动创建默认值

```python
from collections import defaultdict

# 按首字母分组
words = ["apple", "avocado", "banana", "blueberry", "cherry"]
groups = defaultdict(list)
for w in words:
    groups[w[0]].append(w)
# {'a': ['apple', 'avocado'], 'b': ['banana', 'blueberry'], 'c': ['cherry']}

# 对比普通字典，需要先判断键是否存在
plain = {}
for w in words:
    plain.setdefault(w[0], []).append(w)    # 也可以用 setdefault
```

### Counter：计数

```python
from collections import Counter

c = Counter("abracadabra")
print(c)                      # Counter({'a': 5, 'b': 2, 'r': 2, 'c': 1, 'd': 1})
print(c.most_common(2))       # [('a', 5), ('b', 2)]
print(c["z"])                 # 0，不存在的键返回 0，不会报错

# 统计接口返回的状态码
codes = [200, 200, 404, 500, 200]
print(Counter(codes))         # Counter({200: 3, 404: 1, 500: 1})
```

### OrderedDict、ChainMap（了解即可）

3.7 之后普通 dict 已经保序，`OrderedDict` 只在需要 `move_to_end()` 或"顺序参与相等比较"时才用。
`ChainMap` 可以把多个字典叠成一个视图，常用于"命令行参数 > 环境变量 > 默认配置"这种优先级查找。

## 和 JavaScript / TypeScript 对照

| 需求 | JavaScript / TS | Python |
|---|---|---|
| 键值对象 | `{ a: 1 }` | `{"a": 1}`（键要加引号） |
| 取值（不存在） | `obj.k` → `undefined` | `d["k"]` → **KeyError** |
| 安全取值 | `obj.k ?? def` / `obj?.k` | `d.get("k", def)` |
| 键是否存在 | `"k" in obj` | `"k" in d` |
| 合并 | `{...a, ...b}` | `{**a, **b}` 或 `a \| b` |
| 遍历键值 | `Object.entries(o)` | `d.items()` |
| 键可以是对象 | 只有 `Map` 可以 | 任何可哈希对象（如元组） |
| 去重 | `[...new Set(arr)]` | `list(set(lst))` |
| 集合运算 | 手写 filter | `\| & - ^` 运算符 |
| 分组 | `reduce` 或 `Object.groupBy` | `defaultdict(list)` |
| 计数 | `reduce` | `Counter` |

最容易踩的坑：

```python
# 1. 点号访问不适用于字典
user = {"name": "Alice"}
# user.name                # ❌ AttributeError，必须 user["name"]

# 2. 可变默认值共用同一个对象
a = dict.fromkeys(["x", "y"], [])
a["x"].append(1)
print(a)                   # {'x': [1], 'y': [1]}  ❌ 两个键指向同一个列表
a = {k: [] for k in ["x", "y"]}   # ✅ 用推导式，每个键一个新列表

# 3. 遍历时修改字典大小会报错
d = {"a": 1, "b": 2}
# for k in d:
#     del d[k]            # RuntimeError: dictionary changed size during iteration
for k in list(d):          # ✅ 先复制键列表再遍历
    del d[k]
```

## 实战案例

### 案例 1：统计工单状态

```python
from collections import Counter

orders = [
    {"id": 1, "status": "pending"},
    {"id": 2, "status": "done"},
    {"id": 3, "status": "pending"},
    {"id": 4, "status": "done"},
    {"id": 5, "status": "cancelled"},
]

print(Counter(o["status"] for o in orders))
# Counter({'pending': 2, 'done': 2, 'cancelled': 1})
```

### 案例 2：把查询结果按客户分组

```python
from collections import defaultdict

rows = [
    {"customer": "张三", "order": "W001"},
    {"customer": "李四", "order": "W002"},
    {"customer": "张三", "order": "W003"},
]

by_customer = defaultdict(list)
for r in rows:
    by_customer[r["customer"]].append(r["order"])

print(dict(by_customer))
# {'张三': ['W001', 'W003'], '李四': ['W002']}
```

这和 SQL 里的 `GROUP BY customer` 是同一件事，写数据库代码时经常需要在内存里再做一次这样的整理。

### 案例 3：用字典做"分发表"，代替一长串 if-elif

```python
def add(a, b): return a + b
def sub(a, b): return a - b
def mul(a, b): return a * b

OPS = {"add": add, "sub": sub, "mul": mul}

def calc(op: str, a: int, b: int):
    handler = OPS.get(op)
    if handler is None:
        raise ValueError(f"不支持的操作: {op}")
    return handler(a, b)

print(calc("mul", 3, 4))     # 12
```

### 案例 4：两份数据对比，找出新增、删除、保留

```python
old = {"a": 1, "b": 2, "c": 3}
new = {"b": 2, "c": 30, "d": 4}

added   = new.keys() - old.keys()                    # {'d'}
removed = old.keys() - new.keys()                    # {'a'}
changed = {k for k in old.keys() & new.keys() if old[k] != new[k]}   # {'c'}
```

`dict.keys()` 本身就支持集合运算，不需要手动转成 set。

```python
import re
from collections import Counter

def word_count(text):
    words = re.findall(r"[a-z']+", text.lower())
    return dict(Counter(words))

def two_sum(nums, target):
    seen = {}                       # 值 -> 下标
    for i, n in enumerate(nums):
        if target - n in seen:
            return seen[target - n], i
        seen[n] = i
    return None

scores = {"Alice": 90, "Bob": 55, "Carol": 78}
ranked = sorted(scores.items(), key=lambda kv: kv[1], reverse=True)

def find_duplicates(items):
    seen, dup = set(), set()
    for x in items:
        (dup if x in seen else seen).add(x)
    return dup
```
