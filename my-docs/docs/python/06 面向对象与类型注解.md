## 类与对象基础

```python
class User:
    # 类属性：所有实例共享
    species = "human"

    def __init__(self, name: str, age: int):
        # 实例属性：每个实例独立
        self.name = name
        self.age = age

    def greet(self) -> str:
        return f"我是 {self.name}，{self.age} 岁"

u = User("Alice", 25)
print(u.greet())
print(u.name, User.species)
```

要点：

- `__init__` 是初始化方法（类似 JS 的 `constructor`）。
- **每个实例方法的第一个参数必须显式写 `self`**，它就是 JS 里隐含的 `this`。
- Python 没有 `new` 关键字，直接 `User(...)` 就是创建实例。
- 实例属性不需要预先声明，在 `__init__` 里赋值即创建。

### 类属性的陷阱

```python
class Bad:
    items = []                # ❌ 类属性是可变对象，所有实例共用

a, b = Bad(), Bad()
a.items.append(1)
print(b.items)               # [1]

class Good:
    def __init__(self):
        self.items = []       # ✅ 每个实例各自一份
```

### 访问控制：约定而非强制

```python
class Account:
    def __init__(self):
        self.owner = "Alice"      # 公开
        self._balance = 100       # 单下划线：约定为"内部使用"，仍可访问
        self.__secret = "xxx"     # 双下划线：名称改写为 _Account__secret，避免子类冲突
```

Python 没有真正的 `private`，靠命名约定和团队自觉。

## 三种方法

```python
class Order:
    tax_rate = 0.13

    def __init__(self, amount: float):
        self.amount = amount

    # 实例方法：操作某个具体实例
    def total(self) -> float:
        return self.amount * (1 + self.tax_rate)

    # 类方法：第一个参数是类本身，常用来做"备用构造器"
    @classmethod
    def from_dict(cls, data: dict) -> "Order":
        return cls(data["amount"])

    # 静态方法：和类、实例都无关，只是放在类里组织代码
    @staticmethod
    def is_valid_amount(amount: float) -> bool:
        return amount > 0

o = Order.from_dict({"amount": 100})
print(o.total())                       # 113.0
print(Order.is_valid_amount(-1))       # False
```

## 属性：property

```python
class Circle:
    def __init__(self, radius: float):
        self._radius = radius

    @property
    def radius(self) -> float:         # 像属性一样读取：c.radius
        return self._radius

    @radius.setter
    def radius(self, value: float):    # 像属性一样赋值：c.radius = 5
        if value <= 0:
            raise ValueError("半径必须大于 0")
        self._radius = value

    @property
    def area(self) -> float:           # 只读的计算属性
        return 3.14159 * self._radius ** 2

c = Circle(2)
c.radius = 3
print(c.area)
```

对应 JS 的 `get` / `set` 访问器。用 `property` 可以先把字段做成普通属性，
后面需要校验时再改成 property，**调用方的代码不用改**。

## 继承与多态

```python
class Animal:
    def __init__(self, name: str):
        self.name = name

    def speak(self) -> str:
        raise NotImplementedError

class Dog(Animal):
    def __init__(self, name: str, breed: str):
        super().__init__(name)         # 调用父类初始化
        self.breed = breed

    def speak(self) -> str:            # 重写
        return "汪"

class Cat(Animal):
    def speak(self) -> str:
        return "喵"

for a in [Dog("旺财", "柴犬"), Cat("咪咪")]:
    print(a.name, a.speak())           # 多态：同样的调用，不同的行为

print(isinstance(Dog("a", "b"), Animal))   # True
print(Dog.__mro__)                          # 方法解析顺序
```

### 抽象基类

```python
from abc import ABC, abstractmethod

class Repository(ABC):
    @abstractmethod
    def get(self, id: int): ...

    @abstractmethod
    def save(self, obj): ...

# Repository()                 # ❌ TypeError：不能实例化抽象类

class MemoryRepo(Repository):
    def __init__(self):
        self.data = {}
    def get(self, id): return self.data.get(id)
    def save(self, obj): self.data[obj["id"]] = obj
```

### 多重继承与 Mixin（了解）

Python 允许多重继承，实际中最常见的用法是 **Mixin**：把一个小功能做成独立的类，混入需要的类。
继承顺序遵循 MRO（`类.__mro__` 查看），不要写太深的继承链，优先考虑组合。

## 魔术方法

以双下划线开头结尾的方法，让自定义对象支持 Python 的内置语法。

```python
class Vector:
    def __init__(self, x, y):
        self.x, self.y = x, y

    def __repr__(self):                 # 给开发者看：调试、print 列表时显示
        return f"Vector({self.x}, {self.y})"

    def __str__(self):                  # 给用户看：print(v)、str(v)
        return f"({self.x}, {self.y})"

    def __eq__(self, other):            # ==
        return isinstance(other, Vector) and (self.x, self.y) == (other.x, other.y)

    def __add__(self, other):           # +
        return Vector(self.x + other.x, self.y + other.y)

    def __len__(self):                  # len(v)
        return 2

    def __getitem__(self, i):           # v[0]
        return (self.x, self.y)[i]

    def __bool__(self):                 # if v:
        return bool(self.x or self.y)

    def __hash__(self):                 # 定义了 __eq__ 通常要配套定义 __hash__
        return hash((self.x, self.y))

v = Vector(1, 2) + Vector(3, 4)
print(v, repr(v), len(v), v[0])
```

还有两个后面会遇到的：

- `__enter__` / `__exit__`：实现上下文管理器（`with` 语句，04 章讲过）。
- `__call__`：让实例像函数一样被调用 `obj()`。

## dataclass

只存数据的类，手写 `__init__`、`__repr__`、`__eq__` 很啰嗦，`dataclass` 自动生成：

```python
from dataclasses import dataclass, field

@dataclass
class Part:
    part_no: str
    name: str
    price: float = 0.0
    tags: list[str] = field(default_factory=list)   # 可变默认值必须用 default_factory

p = Part("P001", "刹车片", 199.0)
print(p)                       # Part(part_no='P001', name='刹车片', price=199.0, tags=[])
print(p == Part("P001", "刹车片", 199.0))    # True
```

常用选项：

```python
@dataclass(frozen=True)        # 不可变，同时自动生成 __hash__
class Point:
    x: int
    y: int

@dataclass(order=True)         # 自动生成 < > <= >=，按字段顺序比较
class Version:
    major: int
    minor: int

@dataclass(slots=True)         # 3.10+，更省内存、属性访问更快
class Light:
    id: int

from dataclasses import asdict, replace
asdict(p)                      # 转字典
replace(p, price=150.0)        # 复制并修改部分字段（frozen 时很有用）
```

> dataclass 只负责"存"，**不做类型校验**：`Part("P1", "x", price="abc")` 不会报错。
> 需要校验就用下面的 Pydantic。

## 枚举 Enum

```python
from enum import Enum

class OrderStatus(str, Enum):          # 继承 str：值可以直接当字符串用、能被 JSON 序列化
    PENDING = "pending"
    DONE = "done"
    CANCELLED = "cancelled"

s = OrderStatus("done")                # 从值构造，非法值会 ValueError
print(s, s.value, s.name)              # OrderStatus.DONE done DONE
print(s == "done")                     # True（因为继承了 str）
print([x.value for x in OrderStatus])  # 遍历所有成员
```

用枚举代替"魔法字符串"，IDE 能补全，写错值会直接报错。

## 类型注解全解

Python 是动态类型语言，**注解不会在运行时强制检查**，它是给 IDE、mypy/pyright 和
Pydantic、FastAPI 这类库读取的元数据。

### 基础与容器（3.9+ 直接用内置类型）

```python
name: str = "Alice"
age: int = 25
ok: bool = True
scores: list[int] = [1, 2, 3]
point: tuple[int, int] = (1, 2)
mixed: tuple[int, ...] = (1, 2, 3)         # 任意长度的同类型元组
user: dict[str, int] = {"a": 1}
tags: set[str] = {"x"}
```

### 可选与联合（3.10+ 用 `|`）

```python
def find_user(id: int) -> dict | None: ...        # 可能返回 None
def parse(v: int | str) -> int: ...

# 老写法（3.9 及以前），看别人代码时会遇到
from typing import Optional, Union
def f(x: Optional[int] = None): ...               # 等价于 int | None
def g(x: Union[int, str]): ...
```

### Callable、Literal、Any

```python
from typing import Callable, Literal, Any

Handler = Callable[[int, str], bool]               # 接收 (int, str)，返回 bool
def on_event(cb: Callable[[str], None]): ...

def set_mode(mode: Literal["dev", "prod"]): ...    # 只能是这两个字面量
```

### TypedDict：给字典加结构

```python
from typing import TypedDict, NotRequired

class UserDict(TypedDict):
    id: int
    name: str
    email: NotRequired[str]             # 3.11+，可选键

u: UserDict = {"id": 1, "name": "Alice"}
```

### 泛型

```python
from typing import TypeVar, Generic

T = TypeVar("T")

def first(items: list[T]) -> T:
    return items[0]

class Box(Generic[T]):
    def __init__(self, value: T):
        self.value = value

b = Box[int](5)

# 3.12+ 新语法，更简洁
def first2[T](items: list[T]) -> T:
    return items[0]
```

`two/project` 里的 `StandardResponse[UserResponseDto]`，就是一个泛型的响应包装类。

### Protocol：结构化类型（类似 TS 的 interface）

```python
from typing import Protocol

class SupportsClose(Protocol):
    def close(self) -> None: ...

def shutdown(res: SupportsClose):       # 只要对象有 close() 方法就行，不用继承
    res.close()
```

```bash
uv add --dev pyright      # 或 mypy
uv run pyright .
```

## Pydantic（对应 TS 里的 zod）

Pydantic 在**运行时**根据类型注解做校验和转换，是 FastAPI 的核心依赖。

```python
from pydantic import BaseModel, Field, EmailStr, field_validator
```

> `EmailStr` 需要额外安装：`uv add "pydantic[email]"`。

### 定义模型

```python
from datetime import datetime
from pydantic import BaseModel, Field

class CreateUserDto(BaseModel):
    name: str = Field(min_length=1, max_length=50, description="用户姓名")
    age: int = Field(ge=0, le=150)                 # 0 <= age <= 150
    email: str
    tags: list[str] = []                           # Pydantic 会安全处理可变默认值
    nickname: str | None = None

dto = CreateUserDto(name="Alice", age="25", email="a@b.com")
print(dto.age, type(dto.age))        # 25 <class 'int'>   字符串 "25" 被自动转换成了 int
print(dto.model_dump())              # 转字典
print(dto.model_dump_json())         # 转 JSON 字符串
```

### 校验失败

```python
from pydantic import ValidationError

try:
    CreateUserDto(name="", age=-5, email="a@b.com")
except ValidationError as e:
    print(e.errors())                # 列出每个字段的错误位置、原因
```

FastAPI 里请求体校验失败，会自动返回 422 和这样的错误明细，你不用手写任何判断。

### 自定义校验

```python
from pydantic import BaseModel, field_validator, model_validator

class SignUp(BaseModel):
    password: str
    confirm: str

    @field_validator("password")
    @classmethod
    def strong_enough(cls, v: str) -> str:
        if len(v) < 8:
            raise ValueError("密码至少 8 位")
        return v

    @model_validator(mode="after")
    def passwords_match(self):
        if self.password != self.confirm:
            raise ValueError("两次密码不一致")
        return self
```

### 嵌套模型与从 ORM 对象转换

```python
class Address(BaseModel):
    city: str
    street: str

class Customer(BaseModel):
    name: str
    address: Address                 # 嵌套，传字典会被自动转换

c = Customer(name="张三", address={"city": "上海", "street": "xx路"})
print(c.address.city)

# 从 SQLAlchemy 对象等"有属性的对象"转换：打开 from_attributes
from pydantic import ConfigDict

class UserResponseDto(BaseModel):
    model_config = ConfigDict(from_attributes=True)
    id: int
    name: str

# UserResponseDto.model_validate(orm_user)
```

### 配置管理：pydantic-settings

```python
from pydantic_settings import BaseSettings, SettingsConfigDict

class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_file=".env")

settings = Settings()                # 自动读取 .env 与环境变量，并转换成正确类型
```

## 和 TypeScript 对照

| 概念 | TypeScript | Python |
|---|---|---|
| 类型是否运行时存在 | 编译后擦除 | 注解保留，但**默认不检查** |
| 运行时校验 | zod / class-validator | **Pydantic** |
| 接口 | `interface User {}` | `Protocol` / `TypedDict` / `BaseModel` |
| 联合类型 | `string \| number` | `str \| int` |
| 可选 | `name?: string` | `name: str \| None = None` |
| 泛型 | `Box<T>` | `Box[T]`（`Generic[T]`） |
| 字面量类型 | `"a" \| "b"` | `Literal["a", "b"]` |
| 枚举 | `enum` | `Enum` |
| 类的 `this` | 隐式 | 显式 `self` |
| 构造函数 | `constructor` | `__init__` |
| getter/setter | `get x()` | `@property` |
| `static` | `static` | `@staticmethod` / `@classmethod` |
| 只读字段 | `readonly` | `@dataclass(frozen=True)` |
| 对象字面量当数据 | 随手写 | `dict` 或 `dataclass` / `BaseModel` |

最大的心智差别：**TS 的类型错误在编译期暴露，Python 的类型错误要靠 pyright/mypy 检查或 Pydantic 在运行时拦截。**
所以用 Python 做后端，建议在编辑器里打开 pyright 的类型检查。

## 实战案例

### 案例 1：工单模型（dataclass + Enum）

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum

class WorkOrderStatus(str, Enum):
    OPEN = "open"
    IN_PROGRESS = "in_progress"
    CLOSED = "closed"

@dataclass
class WorkOrder:
    id: int
    customer: str
    status: WorkOrderStatus = WorkOrderStatus.OPEN
    created_at: datetime = field(default_factory=datetime.now)

    def start(self):
        if self.status is not WorkOrderStatus.OPEN:
            raise ValueError("只有待处理的工单才能开始")
        self.status = WorkOrderStatus.IN_PROGRESS
```

### 案例 2：用 Pydantic 定义接口的入参与出参

```python
from pydantic import BaseModel, Field, ConfigDict

class CreateWorkOrderDto(BaseModel):
    customer_id: int = Field(gt=0)
    vehicle_vin: str = Field(min_length=17, max_length=17)
    description: str = Field(min_length=1, max_length=500)

class WorkOrderResponseDto(BaseModel):
    model_config = ConfigDict(from_attributes=True)
    id: int
    status: str
    description: str
```

### 案例 3：泛型统一响应包装

```python
from typing import Generic, TypeVar
from pydantic import BaseModel

T = TypeVar("T")

class StandardResponse(BaseModel, Generic[T]):
    code: int = 0
    message: str = "ok"
    data: T | None = None

def success(data: T) -> StandardResponse[T]:
    return StandardResponse(data=data)

print(success({"id": 1}))
```

```python
import re
from dataclasses import dataclass, field
from typing import Callable, Hashable, TypeVar
from pydantic import BaseModel, Field, field_validator

class InsufficientFunds(Exception):
    pass

class Account:
    def __init__(self, balance: float = 0):
        self._balance = balance

    @property
    def balance(self) -> float:
        return self._balance

    def deposit(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("金额必须大于 0")
        self._balance += amount

    def withdraw(self, amount: float) -> None:
        if amount > self._balance:
            raise InsufficientFunds(f"余额 {self._balance}，无法取出 {amount}")
        self._balance -= amount

@dataclass
class Item:
    name: str
    price: float
    qty: int = 1

@dataclass
class Cart:
    items: list[Item] = field(default_factory=list)
    def add(self, item: Item) -> None:
        self.items.append(item)
    def total(self) -> float:
        return sum(i.price * i.qty for i in self.items)
    def __len__(self) -> int:
        return len(self.items)

class RegisterDto(BaseModel):
    username: str = Field(min_length=3, max_length=20)
    password: str = Field(min_length=8)
    age: int = Field(ge=18, le=100)

    @field_validator("password")
    @classmethod
    def has_digit(cls, v: str) -> str:
        if not re.search(r"\d", v):
            raise ValueError("密码必须包含数字")
        return v

T = TypeVar("T")
K = TypeVar("K", bound=Hashable)

def group_by(items: list[T], key: Callable[[T], K]) -> dict[K, list[T]]:
    result: dict[K, list[T]] = {}
    for it in items:
        result.setdefault(key(it), []).append(it)
    return result
```
